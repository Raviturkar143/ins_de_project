---DDL-01 : PRODUCT_MASTER
CREATE TABLE PRODUCT_MASTER
(
    product_id            NUMBER(5)       NOT NULL,
    product_code          VARCHAR2(20)    NOT NULL,
    product_name          VARCHAR2(100)   NOT NULL,
    product_category      VARCHAR2(50)    NOT NULL,
    policy_term_years     NUMBER(2)       NOT NULL,
    premium_frequency     VARCHAR2(20)    NOT NULL,
    product_status        VARCHAR2(20)    DEFAULT 'ACTIVE' NOT NULL,

    created_date          DATE            DEFAULT SYSDATE,
    created_by            VARCHAR2(30)    DEFAULT 'SYSTEM',

    updated_date          DATE,
    updated_by            VARCHAR2(30),

    CONSTRAINT pk_product_master
        PRIMARY KEY (product_id),

    CONSTRAINT uk_product_code
        UNIQUE (product_code),

    CONSTRAINT chk_product_status
        CHECK (product_status IN ('ACTIVE','INACTIVE')),

    CONSTRAINT chk_premium_frequency
        CHECK (premium_frequency IN
            ('MONTHLY',
             'QUARTERLY',
             'HALF_YEARLY',
             'YEARLY',
             'SINGLE'))
);

--DDL-09 09_MONEY_IN_DTL

CREATE TABLE MONEY_IN_DTL
(
    payment_id              NUMBER(12) NOT NULL,

    policy_no               NUMBER(10) NOT NULL,

    installment_no          NUMBER(3),

    payment_type            VARCHAR2(30),

    due_date                DATE,

    payment_date            DATE,

    premium_amount          NUMBER(15,2),

    payment_mode            VARCHAR2(30),

    payment_status          VARCHAR2(20),

    transaction_reference   VARCHAR2(100),

    source_system           VARCHAR2(30),

    batch_id                VARCHAR2(50),

    created_date            DATE DEFAULT SYSDATE,

    created_by              VARCHAR2(30) DEFAULT 'SYSTEM',

    updated_date            DATE,

    updated_by              VARCHAR2(30),

    CONSTRAINT pk_money_in_dtl
        PRIMARY KEY(payment_id),

    CONSTRAINT fk_mid_policy
        FOREIGN KEY(policy_no)
        REFERENCES POLICY(policy_no),

    CONSTRAINT chk_payment_type
        CHECK(payment_type IN
        (
            'FIRST_PREMIUM',
            'RENEWAL_PREMIUM',
            'TOPUP'
        )),

    CONSTRAINT chk_payment_status
        CHECK(payment_status IN
        (
            'PAID',
            'PENDING',
            'FAILED',
            'REFUNDED'
        ))
);
