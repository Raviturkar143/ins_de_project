01_PRODUCT_MASTER.sql
CREATE TABLE PRODUCT_MASTER
(
    product_id            NUMBER(5)       NOT NULL,
    product_code          VARCHAR2(20)    NOT NULL,
    product_name          VARCHAR2(100)   NOT NULL,
    product_category      VARCHAR2(50)    NOT NULL,

    policy_term_years     NUMBER(3),
    premium_frequency     VARCHAR2(20),

    product_status        VARCHAR2(20) DEFAULT 'ACTIVE' NOT NULL,

    source_system         VARCHAR2(30) NOT NULL,
    batch_id              VARCHAR2(50) NOT NULL,

    created_date          DATE DEFAULT SYSDATE,
    created_by            VARCHAR2(30) DEFAULT 'SYSTEM',

    updated_date          DATE,
    updated_by            VARCHAR2(30),

    CONSTRAINT PK_PRODUCT_MASTER
        PRIMARY KEY(product_id),

    CONSTRAINT UK_PRODUCT_CODE
        UNIQUE(product_code),

    CONSTRAINT CHK_PRODUCT_STATUS
        CHECK(product_status IN ('ACTIVE','INACTIVE'))
);


COMMENT ON COLUMN PRODUCT_MASTER.product_id IS
'Internal product identifier';

COMMENT ON COLUMN PRODUCT_MASTER.product_code IS
'Business product code';
02_PRODUCT_COMMISSION_RULE.sql
CREATE TABLE PRODUCT_COMMISSION_RULE
(
    commission_rule_id    NUMBER(10) NOT NULL,

    product_id            NUMBER(5) NOT NULL,

    first_year_pct        NUMBER(5,2) NOT NULL,

    renewal_pct           NUMBER(5,2) NOT NULL,

    effective_from        DATE NOT NULL,

    effective_to          DATE,

    rule_status           VARCHAR2(20)
                          DEFAULT 'ACTIVE' NOT NULL,


    source_system         VARCHAR2(30) NOT NULL,
    batch_id              VARCHAR2(50) NOT NULL,

    created_date          DATE DEFAULT SYSDATE,
    created_by            VARCHAR2(30) DEFAULT 'SYSTEM',

    updated_date          DATE,
    updated_by            VARCHAR2(30),


    CONSTRAINT PK_PRODUCT_COMMISSION_RULE
        PRIMARY KEY(commission_rule_id),


    CONSTRAINT FK_PCR_PRODUCT
        FOREIGN KEY(product_id)
        REFERENCES PRODUCT_MASTER(product_id),


    CONSTRAINT CHK_RULE_STATUS
        CHECK(rule_status IN ('ACTIVE','INACTIVE'))
);


CREATE SEQUENCE PRODUCT_COMM_RULE_SEQ
START WITH 1
INCREMENT BY 1
NOCACHE;

    created_by              VARCHAR2(30) DEFAULT 'SYSTEM',

    updated_by              VARCHAR2(30),

        PRIMARY KEY(customer_policy_id),


    CONSTRAINT FK_CP_CUSTOMER
        FOREIGN KEY(customer_no)
        REFERENCES CUSTOMER(customer_no),


    CONSTRAINT FK_CP_POLICY

        REFERENCES POLICY(policy_no),

09_MONEY_IN_DTL.sql
CREATE TABLE MONEY_IN_DTL
(
    payment_id              NUMBER(12) NOT NULL,


    policy_no               NUMBER(10) NOT NULL,


    installment_no          NUMBER(3),


    payment_type            VARCHAR2(30),


    due_date                DATE,


    payment_date            DATE,


    premium_amount          NUMBER(15,2),


    payment_mode            VARCHAR2(30),


    payment_status          VARCHAR2(20),


    transaction_reference   VARCHAR2(100),



    source_system           VARCHAR2(30) NOT NULL,

    batch_id                VARCHAR2(50) NOT NULL,


    created_date            DATE DEFAULT SYSDATE,

    created_by              VARCHAR2(30) DEFAULT 'SYSTEM',


    updated_date            DATE,

    updated_by              VARCHAR2(30),



    CONSTRAINT PK_MONEY_IN_DTL
        PRIMARY KEY(payment_id),


    CONSTRAINT FK_MID_POLICY
        FOREIGN KEY(policy_no)
        REFERENCES POLICY(policy_no),


    CONSTRAINT CHK_PAYMENT_TYPE
        CHECK(payment_type IN
        (
            'FIRST_PREMIUM',
            'RENEWAL_PREMIUM',
            'TOPUP'
        )),


    CONSTRAINT CHK_PAYMENT_STATUS
        CHECK(payment_status IN
        (
            'PAID',
            'PENDING',
            'FAILED',
            'REFUNDED'
        ))
);



CREATE SEQUENCE MONEY_IN_DTL_SEQ
START WITH 1
INCREMENT BY 1
NOCACHE;
